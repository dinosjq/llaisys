static constexpr unsigned int RADIX_BITS = 8;
static constexpr unsigned int RADIX_SIZE = 1U << RADIX_BITS;
static constexpr unsigned int BLOCK_DIM = 256;
static constexpr unsigned int WARP_SIZE = 32;
static constexpr unsigned int WARP_COUNT = BLOCK_DIM / WARP_SIZE;
static constexpr unsigned int FULL_MASK = 0xFFFFFFFFU;
static constexpr size_t MAX_GRID_X = 2147483647ULL;

// float <---> uint32
__device__ __forceinline__ unsigned int _float_2_uint(float v) {
    unsigned int x = __float_as_uint(v);
    unsigned int mask = (x & 0x80000000U) ? 0xFFFFFFFFU : 0x80000000U;
    return x ^ mask;
}

template <typename T>
__global__ void _topk_kernel(size_t *out_idx,
                             T *out_val,
                             const T *vals,
                             const size_t numel,
                             const size_t k) {
    __shared__ unsigned int warp_hist[WARP_COUNT][RADIX_SIZE];
    __shared__ unsigned int block_hist[RADIX_SIZE];
    __shared__ unsigned int prefix;
    __shared__ unsigned int prefix_mask;
    __shared__ unsigned int rank;
    __shared__ unsigned int greater_reservation;
    __shared__ unsigned int equal_reservation;

    const unsigned int tid = threadIdx.x;
    const unsigned int lane_id = tid & (WARP_SIZE - 1);
    const unsigned int warp_id = tid / WARP_SIZE;
    const size_t batch_id = blockIdx.x;
    const size_t in_offset = batch_id * numel;
    const size_t out_offset = batch_id * k;

    if (tid == 0) {
        prefix = 0;
        prefix_mask = 0;
        rank = static_cast<unsigned int>(k);
    }
    __syncthreads();

#pragma unroll
    for (int pos = 24; pos >= 0; pos -= static_cast<int>(RADIX_BITS)) {
        unsigned int *flat_hist = &warp_hist[0][0];
        for (unsigned int offset = tid; offset < WARP_COUNT * RADIX_SIZE; offset += BLOCK_DIM) {
            flat_hist[offset] = 0;
        }
        __syncthreads();

        // A warp iterates whole 32-element chunks so every lane reaches each
        // full-mask ballot, including the final partial chunk.
        for (size_t base = warp_id * WARP_SIZE; base < numel; base += BLOCK_DIM) {
            const size_t element = base + lane_id;
            unsigned int key = 0;
            unsigned int digit = 0;
            bool candidate = false;
            if (element < numel) {
                const float value = llaisys::utils::nvidia::cast<float>(vals[in_offset + element]);
                key = _float_2_uint(value);
                candidate = (key & prefix_mask) == prefix;
                digit = (key >> pos) & (RADIX_SIZE - 1);
            }

            const unsigned int active = __ballot_sync(FULL_MASK, candidate);
            if (candidate) {
                const unsigned int peers = __match_any_sync(active, digit);
                const unsigned int leader = static_cast<unsigned int>(__ffs(peers) - 1);
                if (lane_id == leader) {
                    atomicAdd(&warp_hist[warp_id][digit], static_cast<unsigned int>(__popc(peers)));
                }
            }
        }

        __syncthreads();
        if (tid < RADIX_SIZE) {
            unsigned int count = 0;
#pragma unroll
            for (unsigned int warp = 0; warp < WARP_COUNT; ++warp) {
                count += warp_hist[warp][tid];
            }
            block_hist[tid] = count;
        }
        __syncthreads();

        if (tid == 0) {
            for (int digit = static_cast<int>(RADIX_SIZE) - 1; digit >= 0; --digit) {
                const unsigned int count = block_hist[digit];
                if (count >= rank) {
                    prefix |= static_cast<unsigned int>(digit) << pos;
                    break;
                }
                rank -= count;
            }
            prefix_mask = ~((1U << pos) - 1U);
        }
        __syncthreads();
    }

    if (tid == 0) {
        greater_reservation = 0;
        equal_reservation = 0;
    }
    __syncthreads();

    const unsigned int greater_quota = static_cast<unsigned int>(k) - rank;
    const unsigned int equal_quota = rank;

    for (size_t base = warp_id * WARP_SIZE; base < numel; base += BLOCK_DIM) {
        const size_t element = base + lane_id;
        unsigned int key = 0;
        const bool valid = element < numel;
        if (valid) {
            const float value = llaisys::utils::nvidia::cast<float>(vals[in_offset + element]);
            key = _float_2_uint(value);
        }

        const bool is_greater = valid && key > prefix;
        const bool is_equal = valid && key == prefix;
        const unsigned int greater_ballot = __ballot_sync(FULL_MASK, is_greater);
        const unsigned int equal_ballot = __ballot_sync(FULL_MASK, is_equal);

        if (is_greater) {
            const unsigned int leader = static_cast<unsigned int>(__ffs(greater_ballot) - 1);
            unsigned int reservation = 0;
            if (lane_id == leader) {
                reservation = atomicAdd(&greater_reservation, static_cast<unsigned int>(__popc(greater_ballot)));
            }
            reservation = __shfl_sync(greater_ballot, reservation, leader);
            const unsigned int local_rank = static_cast<unsigned int>(__popc(greater_ballot & ((1U << lane_id) - 1U)));
            const unsigned int output = reservation + local_rank;
            if (output < greater_quota) {
                out_idx[out_offset + output] = element;
                out_val[out_offset + output] = vals[in_offset + element];
            }
        }

        if (is_equal) {
            const unsigned int leader = static_cast<unsigned int>(__ffs(equal_ballot) - 1);
            unsigned int reservation = 0;
            if (lane_id == leader) {
                reservation = atomicAdd(&equal_reservation, static_cast<unsigned int>(__popc(equal_ballot)));
            }
            reservation = __shfl_sync(equal_ballot, reservation, leader);
            const unsigned int local_rank = static_cast<unsigned int>(__popc(equal_ballot & ((1U << lane_id) - 1U)));
            const unsigned int output = reservation + local_rank;
            if (output < equal_quota) {
                const size_t destination = out_offset + greater_quota + output;
                out_idx[destination] = element;
                out_val[destination] = vals[in_offset + element];
            }
        }

        __syncwarp(FULL_MASK);
    }
}

template <typename T>
void _topk_launch(std::byte *out_idx,
                  std::byte *out_val,
                  const std::byte *vals,
                  size_t batch_size,
                  size_t numel,
                  size_t k) {
    if (batch_size == 0 || k == 0) {
        return;
    }
    if (batch_size > MAX_GRID_X || numel == 0 || k > numel
        || numel > std::numeric_limits<unsigned int>::max()
        || k > std::numeric_limits<unsigned int>::max()) {
        throw std::invalid_argument(
            "topk requires batch_size <= 2^31-1 and "
            "0 < k <= numel <= UINT_MAX");
    }

    size_t *d_out_idx = reinterpret_cast<size_t *>(out_idx);
    T *d_out_val = reinterpret_cast<T *>(out_val);
    const T *d_vals = reinterpret_cast<const T *>(vals);
    
    dim3 blockDim(BLOCK_DIM);
    dim3 gridDim(batch_size);

    _topk_kernel<<<gridDim, blockDim>>>(d_out_idx, d_out_val, d_vals, numel, k);

    CUDA_CHECK(cudaGetLastError());
}

namespace llaisys::ops::nvidia {
void topk(std::byte *out_idx, std::byte *out_val, const std::byte *vals, llaisysDataType_t type, size_t batch_size, size_t numel, size_t k){
    switch (type) {
    case LLAISYS_DTYPE_F32:
        return _topk_launch<float>(out_idx, out_val, vals, batch_size, numel, k);
    case LLAISYS_DTYPE_BF16:
        return _topk_launch<llaisys::bf16_t>(out_idx, out_val, vals, batch_size, numel, k);
    case LLAISYS_DTYPE_F16:
        return _topk_launch<llaisys::fp16_t>(out_idx, out_val, vals, batch_size, numel, k);
    default:
        EXCEPTION_UNSUPPORTED_DATATYPE(type);
    }
}
} // namespace llaisys::ops::nvidia
